SELECT
    c.login,
    COUNT(o.*) AS in_delivery_count
FROM "Couriers" AS c
JOIN "Orders" AS o ON c.id = o."courierId"
WHERE o."inDelivery" = TRUE
GROUP BY c.id, c.login
ORDER BY c.login;

SELECT
    o.track,
    CASE
        WHEN o.finished = TRUE    THEN 2
        WHEN o.cancelled = TRUE  THEN -1
        WHEN o."inDelivery" = TRUE THEN 1
        ELSE 0
    END AS status
FROM "Orders" AS o
ORDER BY o.track;
